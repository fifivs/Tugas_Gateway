Pseudocode Algoritma Backtracking 

function backtracking_route(app_id, data, candidates, index=0, trace=[]):
    // BASE CASE: semua kandidat sudah dicoba
    if index >= len(candidates):
        return "GAGAL — semua route exhausted"

    candidate = candidates[index]

    try:
        result = send_request(candidate.url, data)
        return "SUKSES" + trace     // Solusi ditemukan!

    except (ConnectError, Timeout):
        trace.append("GAGAL → BACKTRACK")
        // RECURSIVE CALL — backtrack ke kandidat berikutnya
        return backtracking_route(app_id, data, candidates, index+1, trace)
